{$CLEO .cs}
{$USE CLEO}
{$VERSION CLEO_5.4}
script_name 'INFAMMO'

// ================================================================================
// MUNIÇÃO INFINITA SEM RECARREGAR (NO RELOAD / INF AMMO)
// Compatível com: GTA San Andreas v1.0 US / CLEO 5.4+ / Sanny Builder 4
// HotKey: INSERT (VK 45) - Toggle ON/OFF
// ================================================================================

var
    0@ : Int  // Status do mod (ON=1, OFF=0)
    1@ : Int  // Ponteiro do jogador (CPed)
    2@ : Int  // Slot de arma atual
    4@ : Int  // Endereço da arma atual na memória
    7@ : Int  // Flag de controle de keypress (debounce)
    9@ : Int  // Variável auxiliar de cálculo
end

// Inicialização segura
0@ = 0
7@ = 0

while true
    wait 0

    // ================================================================
    // HOTKEY CHECK - Tecla INSERT (VK 45)
    // ================================================================
    if is_key_pressed 45
    then
        if 7@ == 0
        then
            7@ = 1

            if 0@ == 0
            then
                0@ = 1
                00BC: print_formatted_now "Municao Infinita: ATIVADO" time 2000 flag 1
            else
                0@ = 0
                00BC: print_formatted_now "Municao Infinita: DESATIVADO" time 2000 flag 1
            end
        end
    else
        7@ = 0
    end

    // ================================================================
    // LÓGICA PRINCIPAL DO MOD
    // ================================================================
    1@ = read_memory 0xB6F5F0 size 4 virtual_protect 0

    if 1@ > 0
    then
        // Lê o slot da arma atual (CPed + 0x718)
        9@ = 1@ + 0x718
        2@ = read_memory 9@ size 1 virtual_protect 0

        if 2@ >= 0
        then
            if 2@ <= 12
            then
                // No GTA SA, o array CWeapon de 13 slots fica DIRETO em CPed + 0x5A0 (28 bytes por slot)
                9@ = 2@ * 28
                4@ = 1@ + 0x5A0
                4@ += 9@

                if 0@ == 1
                then
                    // Trava o pente de munição (m_nAmmoInClip em +0x08)
                    9@ = 4@ + 0x08
                    write_memory 9@ size 4 value 9999 virtual_protect 0

                    // Trava a munição total (m_nTotalAmmo em +0x0C)
                    9@ = 4@ + 0x0C
                    write_memory 9@ size 4 value 9999 virtual_protect 0
                end
            end
        end
    end
end
